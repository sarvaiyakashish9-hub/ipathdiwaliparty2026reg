function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  if (data.action === "delete") {
    var rows = sheet.getDataRange().getValues();
    for (var i = rows.length - 1; i >= 1; i--) {
      if (String(rows[i][0]) === String(data.id)) {
        sheet.deleteRow(i + 1);
      }
    }
    return ContentService.createTextOutput(JSON.stringify({status: "deleted"})).setMimeType(ContentService.MimeType.JSON);
  }
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["ID", "Name", "Department", "Attending", "Spouse", "Has Kids", "Bringing Kids", "Kids Count", "Total Headcount", "Timestamp"]);
  }
  sheet.appendRow([data.id, data.name, data.department, data.attending, data.spouse, data.hasKids, data.bringingKids, data.kidsCount, data.totalHeadcount, data.createdAt]);
  return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var values = sheet.getDataRange().getValues();
  if (values.length <= 1) return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  var headers = values[0];
  var rows = [];
  for (var i = 1; i < values.length; i++) {
    var row = values[i];
    rows.push({
      id: row[0],
      name: row[1],
      department: row[2],
      attending: row[3],
      spouse: row[4],
      hasKids: row[5],
      bringingKids: row[6],
      kidsCount: Number(row[7]) || 0,
      totalHeadcount: Number(row[8]) || 0,
      createdAt: row[9]
    });
  }
  return ContentService.createTextOutput(JSON.stringify(rows)).setMimeType(ContentService.MimeType.JSON);
}
